**Goal:** Wire the SDK's mention surface into the demo — a `@username` typeahead in the comment box and new-post form, and clickable `@username` links in displayed posts/comments that open the profile overlay.

**Architecture:** Two new components. `MentionTextarea` (compose) wraps a controlled textarea + the headless `useUserMentions` hook + a suggestions dropdown, exposing accumulated `Mention[]` to parents via an imperative ref handle. `MentionText` (display) linkifies `@username` tokens against the server-persisted `mentions` array on the entity/comment. Composers thread `mentions` into their existing create calls.

**Tech Stack:** Vite + React + TypeScript, `@agora-sdk/core` (`useUserMentions`), `@agora-sdk/react-js`. No test framework — the static gate is `pnpm build`; verification is manual click-through per repo convention.

```tsx
import { forwardRef, useImperativeHandle, useRef, useState } from "react";
import { useUserMentions } from "@agora-sdk/core";
import { track } from "./analytics";

// A drop-in replacement for a <textarea> that adds an @username mention typeahead. It stays
// controlled (the parent owns value/onChange) and hides all the mention mechanics: it tracks the
// caret, drives the SDK's headless useUserMentions hook (debounced GET /users/suggestions), and
// renders a suggestions dropdown beneath the field. On pick, the hook rewrites the "@partial" in the
// text to "@username " and records a Mention. The parent reads the accumulated mentions at submit via
// the imperative handle and clears them after.
//
// useUserMentions comes from @agora-sdk/core (react-js doesn't re-export it) — the demo's one direct
// core import. Safe: a single deduped core instance shares the AgoraProvider context.

export interface MentionTextareaHandle {
  getMentions: () => any[];
  reset: () => void;
}

interface MentionTextareaProps {
  value: string;
  onChange: (v: string) => void;
  placeholder?: string;
  rows?: number;
  disabled?: boolean;
  analyticsTarget: "comment" | "entity";
  onKeyDown?: (e: React.KeyboardEvent<HTMLTextAreaElement>) => void;
}

const MentionTextarea = forwardRef<MentionTextareaHandle, MentionTextareaProps>(
  function MentionTextarea(
    { value, onChange, placeholder, rows = 3, disabled, analyticsTarget, onKeyDown },
    ref,
  ) {
    const textareaRef = useRef<HTMLTextAreaElement>(null);
    const [cursorPosition, setCursorPosition] = useState(0);
    const [isSelectionActive, setIsSelectionActive] = useState(false);

    // minChars: 2 → suggestions after "@" + 2 chars (SDK default is 3). Debounce left at the SDK
    // default (1000ms) to avoid hammering /users/suggestions.
    const {
      isMentionActive,
      loading,
      mentionSuggestions,
      handleMentionClick,
      mentions,
      resetMentions,
    } = useUserMentions({
      content: value,
      setContent: onChange,
      focus: () => textareaRef.current?.focus(),
      cursorPosition,
      isSelectionActive,
      minChars: 2,
    }) as any;

    useImperativeHandle(
      ref,
      () => ({ getMentions: () => mentions, reset: () => resetMentions() }),
      [mentions, resetMentions],
    );

    // Re-read caret + selection from the textarea after anything that can move them, so the hook
    // knows which word the cursor is in.
    const syncCaret = () => {
      const el = textareaRef.current;
      if (!el) return;
      setCursorPosition(el.selectionStart ?? 0);
      setIsSelectionActive(el.selectionStart !== el.selectionEnd);
    };

    const pick = (user: any) => {
      handleMentionClick(user);
      track("insert_mention", { target: analyticsTarget });
    };

    return (
      <div style={{ position: "relative" }}>
        <textarea
          ref={textareaRef}
          placeholder={placeholder}
          rows={rows}
          disabled={disabled}
          value={value}
          onChange={(e) => { onChange(e.target.value); syncCaret(); }}
          onKeyUp={syncCaret}
          onClick={syncCaret}
          onSelect={syncCaret}
          onKeyDown={onKeyDown}
        />
        {isMentionActive && (loading || mentionSuggestions.length > 0) && (
          <div
            className="col"
            style={{
              position: "absolute", left: 0, right: 0, top: "100%", zIndex: 50,
              gap: 0, marginTop: 2, border: "1px solid var(--border)", borderRadius: 8,
              background: "var(--panel)", boxShadow: "0 10px 30px rgba(0,0,0,0.45)",
              maxHeight: 220, overflowY: "auto",
            }}
          >
            {mentionSuggestions.length === 0 ? (
              <div className="muted" style={{ padding: "8px 10px" }}>searching…</div>
            ) : (
              mentionSuggestions.map((u: any) => (
                <button
                  key={u.id}
                  type="button"
                  className="row"
                  // onMouseDown + preventDefault so picking a suggestion doesn't blur the textarea
                  // (which would drop the selection before the handler runs) — the typeahead idiom.
                  onMouseDown={(e) => { e.preventDefault(); pick(u); }}
                  style={{
                    gap: 8, alignItems: "center", padding: "6px 10px", border: 0,
                    background: "transparent", width: "100%", textAlign: "left", cursor: "pointer",
                  }}
                >
                  {u.avatar ? (
                    <img src={u.avatar} alt="" style={{ width: 20, height: 20, borderRadius: "50%", objectFit: "cover" }} />
                  ) : (
                    <span style={{ width: 20, height: 20, borderRadius: "50%", background: "var(--border)", display: "inline-block" }} />
                  )}
                  <span><strong>@{u.username}</strong>{u.name ? <span className="muted"> · {u.name}</span> : null}</span>
                </button>
              ))
            )}
          </div>
        )}
      </div>
    );
  },
);

export default MentionTextarea;
```

Create `src/MentionText.tsx`:

```tsx
import type { ReactNode } from "react";
import { useProfileViewer } from "./ProfileViewerContext";

// Renders content with @username tokens linkified. A token whose username is in the object's
// server-persisted `mentions` array (entity.mentions / comment.mentions) becomes clickable and opens
// the profile overlay; unknown @tokens and all other text stay plain. Keeps the caller's className
// (e.g. "prewrap") on the container so whitespace/newlines render exactly as before.
export default function MentionText({
  content,
  mentions,
  className,
}: {
  content?: string | null;
  mentions?: any[];
  className?: string;
}) {
  const { openProfile } = useProfileViewer();
  if (!content) return null;

  // Exact username → mention (user mentions only).
  const byUsername = new Map<string, any>();
  for (const m of mentions ?? []) {
    if (m?.type === "user" && m.username) byUsername.set(m.username, m);
  }

  // Split into @token / plain-text runs in order. Same charset the compose hook allows (@[\w.]+).
  const parts: ReactNode[] = [];
  const re = /@([\w.]+)/g;
  let last = 0;
  let key = 0;
  let match: RegExpExecArray | null;
  while ((match = re.exec(content)) !== null) {
    if (match.index > last) parts.push(content.slice(last, match.index));
    const username = match[1];
    const mention = byUsername.get(username);
    if (mention) {
      parts.push(
        <span
          key={key++}
          onClick={() => openProfile(mention.id)}
          style={{ color: "var(--text)", textDecoration: "underline", textUnderlineOffset: 3, cursor: "pointer", fontWeight: 500 }}
        >
          @{username}
        </span>,
      );
    } else {
      parts.push(match[0]);
    }
    last = match.index + match[0].length;
  }
  if (last < content.length) parts.push(content.slice(last));

  return <div className={className}>{parts}</div>;
}
```
